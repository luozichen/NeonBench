    def __init__(self, ndim, bias=True):
        super().__init__()
        self.weight = nn.Parameter(torch.ones(ndim))
        self.bias = nn.Parameter(torch.zeros(ndim)) if bias else None
    def forward(self, input):
        return F.layer_norm(input, self.weight.shape, self.weight, self.bias, 1e-5)

KV is compressed into a low-rank latent before up-projecting to K and V."""
import torch
import torch.nn as nn
from torch.nn import functional as F

class RMSNorm(nn.Module):
    def __init__(self, dim, eps=1e-6):
        super().__init__()
        self.eps = eps
        self.weight = nn.Parameter(torch.ones(dim))
    def forward(self, x):
        return x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.eps) * self.weight

def apply_rotary_emb(x, freqs_cos, freqs_sin):
    d = x.shape[-1]
    x_even, x_odd = x[..., :d:2], x[..., 1:d:2]
    cos = freqs_cos[:x.shape[1]].view(1, x.shape[1], 1, -1)
    sin = freqs_sin[:x.shape[1]].view(1, x.shape[1], 1, -1)
    return torch.cat([x_even * cos - x_odd * sin, x_even * sin + x_odd * cos], dim=-1)

class MLAttention(nn.Module):
    """Multi-head Latent Attention: compress KV through a shared bottleneck."""
    def __init__(self, config):
        super().__init__()
        self.n_head = config['n_head']
        self.head_dim = config['d_model'] // config['n_head']
        d_model = config['d_model']
        d_latent = config.get('d_latent', d_model // 2)

        # Q projection (standard)
        self.q_proj = nn.Linear(d_model, d_model, bias=False)
        self.q_norm = RMSNorm(self.head_dim)

        # KV compression: down-project to latent, then up-project to K and V
        self.kv_down = nn.Linear(d_model, d_latent, bias=False)
        self.k_up = nn.Linear(d_latent, d_model, bias=False)
        self.v_up = nn.Linear(d_latent, d_model, bias=False)
        self.k_norm = RMSNorm(self.head_dim)

        self.c_proj = nn.Linear(d_model, d_model, bias=False)

    def forward(self, x, freqs_cos, freqs_sin):
        B, T, C = x.shape
        q = self.q_proj(x)

        # Compress KV through latent bottleneck
        c_kv = self.kv_down(x)          # (B, T, d_latent)
        k = self.k_up(c_kv)             # (B, T, d_model)
        v = self.v_up(c_kv)

Formula: Output_i = I_i ⊙ Σ_j(A_ij V_j)
Intent is projected per-head (same as Q, K, V)."""
import torch
import torch.nn as nn
from torch.nn import functional as F

class RMSNorm(nn.Module):
    def __init__(self, dim, eps=1e-6):
        super().__init__()
        self.eps = eps
        self.weight = nn.Parameter(torch.ones(dim))
    def forward(self, x):
        return x * torch.rsqrt(x.pow(2).mean(-1, keepdim=True) + self.eps) * self.weight

def apply_rotary_emb(x, freqs_cos, freqs_sin):
    d = x.shape[-1]
    x_even, x_odd = x[..., :d:2], x[..., 1:d:2]
    cos = freqs_cos[:x.shape[1]].view(1, x.shape[1], 1, -1)
    sin = freqs_sin[:x.shape[1]].view(1, x.shape[1], 1, -1)
    return torch.cat([x_even * cos - x_odd * sin, x_even * sin + x_odd * cos], dim=-1)

class IntentAttention(nn.Module):
    """QKVI Attention: Q, K, V, I all projected per-head.
    Result gating: Output = I ⊙ Attn(Q, K, V)"""
    def __init__(self, config):
        super().__init__()
        self.n_head = config['n_head']
        self.head_dim = config['d_model'] // config['n_head']
        d_model = config['d_model']

        self.c_attn = nn.Linear(d_model, 4 * d_model, bias=False)  # Q, K, V, I
        self.q_norm = RMSNorm(self.head_dim)
        self.k_norm = RMSNorm(self.head_dim)
        self.c_proj = nn.Linear(d_model, d_model, bias=False)
